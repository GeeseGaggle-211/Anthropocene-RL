"""Own-knowledge fixed markets and a protected, explicit trade-head curriculum."""

import hashlib

import numpy as np
import torch

from anthropocene_rl.autonomous_v4.policy import Action

TRADE_HEADS = ("enabled", "partner", "give", "want", "lot", "price")


def own_address(transport, observer, target):
    result = transport._request(
        dict(op="autonomous6_quote_address", observer=observer, target=target)
    )
    if result.get("metadataVersion") != "own-known-address-v1":
        raise ValueError("Known-address metadata version mismatch")
    value = result.get("ordinal")
    if type(value) is not int or not -1 <= value < 8:
        raise ValueError("Invalid known-contact ordinal")
    return value


def market_peer(obs, peer, focal, transport, entry):
    # Never receives a focal observation/action. A missing known customer does
    # not authorize silently choosing another legal first contact.
    ordinal = own_address(transport, peer, focal)
    give = 9 - entry["market_focal_give"]
    want = entry["market_focal_give"]
    workforce = np.zeros(13)
    workforce[0] = 0.6 if obs["WorkforceMask"][1] else 1
    workforce[1] = 1 - workforce[0]
    activities = obs["ActivityMask"].astype(float)
    activities /= np.maximum(1, activities.sum(-1, keepdims=True))
    activities[0] = 0
    activities[0, 0] = 1
    result = dict(workforce=workforce, activities=activities, controls=np.zeros(3), movement=0)
    if (
        ordinal >= 0
        and obs["CounterpartyMask"][ordinal]
        and obs["GoodMask"][give]
        and obs["GoodMask"][want]
        and obs["Stocks"][give] >= 1e-6
    ):
        quantity = max(1e-6, min(1e6, obs["Stocks"][give] * 0.1))
        result["quote"] = dict(
            Counterparty=ordinal,
            Give=give,
            Want=want,
            GiveQuantity=float(quantity),
            WantQuantity=float(quantity),
        )
    return result


def trade_target(obs, ordinal, give):
    want = 9 - give
    if give not in (3, 6) or not -1 <= ordinal < 8:
        raise ValueError("Invalid public market target")
    enabled = (
        ordinal >= 0
        and obs["CounterpartyMask"][ordinal]
        and obs["GoodMask"][give]
        and obs["GoodMask"][want]
        and obs["Stocks"][give] >= 1e-6
    )
    return [1, ordinal, give, want, 1, 1] if enabled else [0] * 6


def protected_warmup_state(trainer):
    """All tensors other than six named trade heads, plus complete PPO state."""
    digest = hashlib.sha256()

    def add(value):
        if isinstance(value, torch.Tensor):
            digest.update(str((value.dtype, value.shape)).encode())
            digest.update(value.detach().cpu().numpy().tobytes())
        elif isinstance(value, dict):
            for key in sorted(value, key=repr):
                digest.update(repr(key).encode())
                add(value[key])
        elif isinstance(value, (list, tuple)):
            for item in value:
                add(item)
        else:
            digest.update(repr(value).encode())

    for name, value in trainer.model.state_dict().items():
        if not any(name.startswith(f"heads.{head}.") for head in TRADE_HEADS):
            digest.update(name.encode())
            add(value)
    add(trainer.optimizer.state_dict())
    add(trainer.sampler.get_state())
    add(trainer.shuffle.get_state())
    add(
        (trainer.episodes, trainer.transitions, trainer.updates, trainer.config, trainer.provenance)
    )
    add([(n, p.requires_grad) for n, p in trainer.model.named_parameters()])
    return digest.hexdigest()


def warmup_trade(trainer, observations, targets, *, seed, steps=64, lr=0.003):
    if not 1 <= steps <= 64 or lr != 0.003:
        raise ValueError("Frozen trade warmup envelope")
    if targets.shape != (len(observations["x"]), 6) or not len(targets):
        raise ValueError("Empty/invalid native trade corpus")
    before = protected_warmup_state(trainer)
    flags = {n: p.requires_grad for n, p in trainer.model.named_parameters()}
    parameters = [
        p
        for n, p in trainer.model.named_parameters()
        if any(n.startswith(f"heads.{head}.") for head in TRADE_HEADS)
    ]
    original = torch.cat([p.detach().flatten() for p in parameters])
    generator = torch.Generator().manual_seed(seed)
    with torch.no_grad():
        sampled = trainer.model(observations, generator=generator)[0]
    action = Action(
        sampled.workforce, sampled.activities, sampled.controls, sampled.movement, targets
    )
    optimizer = torch.optim.Adam(parameters, lr=lr, eps=1e-5)
    try:
        for name, p in trainer.model.named_parameters():
            p.requires_grad_(any(name.startswith(f"heads.{head}.") for head in TRADE_HEADS))
        with torch.no_grad():
            start = float(trainer.model(observations, action)[1].mean())
        for _ in range(steps):
            loss = -trainer.model(observations, action)[1].mean()
            if not torch.isfinite(loss):
                raise ValueError("Nonfinite warmup likelihood")
            optimizer.zero_grad(set_to_none=True)
            loss.backward()
            torch.nn.utils.clip_grad_norm_(parameters, 0.5, error_if_nonfinite=True)
            optimizer.step()
        with torch.no_grad():
            end = float(trainer.model(observations, action)[1].mean())
    finally:
        for name, p in trainer.model.named_parameters():
            p.requires_grad_(flags[name])
        for p in parameters:
            p.grad = None
    after = protected_warmup_state(trainer)
    if before != after:
        raise ValueError("Trade warmup changed protected PPO/model state")
    return dict(
        optimizer_steps=steps,
        native_resolves=0,
        retained_ppo_transitions=0,
        joint_logp_fixed_nontrade_before=start,
        joint_logp_fixed_nontrade_after=end,
        trade_head_step_l2=float(
            (torch.cat([p.detach().flatten() for p in parameters]) - original).norm()
        ),
        protected_state_before=before,
        protected_state_after=after,
        temporary_optimizer_discarded=True,
    )
